```rust
use std::fs;
use std::io::Read;
use std::path::{Component, Path, PathBuf};

const ONE_MB: u64 = 1024 * 1024;
const FIVE_MB: u64 = 5 * ONE_MB;
const SNIFF: usize = 8 * 1024;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum FileError {
    OutsideWorkspace,
    NotFound,
    Io(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ReadOutcome {
    Text { bytes: Vec<u8>, highlight: bool },
    TooLarge { bytes: u64 },
    Binary,
}

pub fn read_file(root: &Path, absolute: PathBuf) -> Result<ReadOutcome, FileError> {
    let absolute = normalize_inside(root, &absolute)?;
    let meta = fs::metadata(&absolute).map_err(|e| {
        if e.kind() == std::io::ErrorKind::NotFound {
            FileError::NotFound
        } else {
            FileError::Io(e.to_string())
        }
    })?;
    if !meta.is_file() {
        return Err(FileError::Io("not a file".into()));
    }
    let len = meta.len();
    if len > FIVE_MB {
        return Ok(ReadOutcome::TooLarge { bytes: len });
    }
    let mut file = fs::File::open(&absolute).map_err(|e| FileError::Io(e.to_string()))?;
    let mut bytes = Vec::new();
    file.read_to_end(&mut bytes)
        .map_err(|e| FileError::Io(e.to_string()))?;
    let sniff = &bytes[..bytes.len().min(SNIFF)];
    if sniff.contains(&0) || std::str::from_utf8(&bytes).is_err() {
        return Ok(ReadOutcome::Binary);
    }
    Ok(ReadOutcome::Text {
        bytes,
        highlight: len <= ONE_MB,
    })
}

pub fn normalize_inside(root: &Path, candidate: &Path) -> Result<PathBuf, FileError> {
    let root = root.canonicalize().map_err(|e| FileError::Io(e.to_string()))?;
    let candidate = if candidate.exists() {
        candidate
            .canonicalize()
            .map_err(|e| FileError::Io(e.to_string()))?
    } else {
        lexical_normalize(candidate)
    };
    if candidate.starts_with(&root) {
        Ok(candidate)
    } else {
        Err(FileError::OutsideWorkspace)
    }
}

fn lexical_normalize(path: &Path) -> PathBuf {
    let mut out = PathBuf::new();
    for c in path.components() {
        match c {
            Component::CurDir => {}
            Component::ParentDir => {
                out.pop();
            }
            other => out.push(other.as_os_str()),
        }
    }
    out
}
```

```rust
pub fn write_file(root: &Path, absolute: &Path, text: &str) -> Result<(), FileError> {
    let absolute = normalize_inside(root, absolute)?;
    if absolute.is_dir() {
        return Err(FileError::Io("refusing to write a directory".into()));
    }
    let tmp = absolute.with_extension("vela-tmp");
    fs::write(&tmp, text.as_bytes()).map_err(|e| FileError::Io(e.to_string()))?;
    fs::rename(&tmp, &absolute).map_err(|e| {
        let _ = fs::remove_file(&tmp);
        FileError::Io(e.to_string())
    })
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EntryKind {
    File,
    Dir,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub name: String,
    pub path: String,
    pub kind: EntryKind,
}

const SKIP_DIRS: &[&str] = &[".git", "node_modules", "target", "dist"];

pub fn list_children(root: &Path, directory: &Path) -> Result<Vec<Entry>, FileError> {
    let directory = normalize_inside(root, directory)?;
    let mut entries = Vec::new();
    let rd = fs::read_dir(&directory).map_err(|e| FileError::Io(e.to_string()))?;
    for item in rd {
        let item = item.map_err(|e| FileError::Io(e.to_string()))?;
        let file_type = item.file_type().map_err(|e| FileError::Io(e.to_string()))?;
        if file_type.is_symlink() {
            continue;
        }
        let name = item.file_name().to_string_lossy().into_owned();
        if file_type.is_dir() && SKIP_DIRS.contains(&name.as_str()) {
            continue;
        }
        let rel = item
            .path()
            .strip_prefix(root.canonicalize().map_err(|e| FileError::Io(e.to_string()))?)
            .map_err(|_| FileError::OutsideWorkspace)?;
        let path = rel
            .components()
            .map(|c| c.as_os_str().to_string_lossy())
            .collect::<Vec<_>>()
            .join("/");
        entries.push(Entry {
            name,
            path,
            kind: if file_type.is_dir() {
                EntryKind::Dir
            } else {
                EntryKind::File
            },
        });
    }
    entries.sort_by(|a, b| a.name.cmp(&b.name));
    Ok(entries)
}
```

```rust
use serde::Serialize;
use std::path::PathBuf;
use std::sync::atomic::{AtomicU64, Ordering};
use tauri::State;

struct AppState {
    generation: AtomicU64,
    root: std::sync::Mutex<Option<PathBuf>>,
}

#[derive(Serialize)]
#[serde(tag = "kind", rename_all = "snake_case")]
enum ReadDto {
    Text { bytes: Vec<u8>, highlight: bool },
    TooLarge { bytes: u64 },
    Binary,
}

#[derive(Serialize)]
struct OpenDto {
    generation: u64,
    walk_id: u64,
    root_entries: Vec<files::Entry>,
    watch_ok: bool,
}

#[tauri::command]
fn open_workspace(state: State<AppState>, path: String) -> Result<OpenDto, String> {
    let root = PathBuf::from(&path);
    if !root.is_dir() {
        return Err("not a directory".into());
    }
    let entries = files::list_children(&root, &root).map_err(|e| format!("{e:?}"))?;
    *state.root.lock().map_err(|e| e.to_string())? = Some(root);
    let generation = state.generation.fetch_add(1, Ordering::SeqCst) + 1;
    Ok(OpenDto {
        generation,
        walk_id: 0,
        root_entries: entries,
        watch_ok: false,
    })
}

#[tauri::command]
fn list_children(state: State<AppState>, generation: u64, path: String) -> Result<Vec<files::Entry>, String> {
    let current = state.generation.load(Ordering::SeqCst);
    if generation != current {
        return Err("stale workspace".into());
    }
    let root = state
        .root
        .lock()
        .map_err(|e| e.to_string())?
        .clone()
        .ok_or("no workspace")?;
    let absolute = root.join(path);
    files::list_children(&root, &absolute).map_err(|e| format!("{e:?}"))
}

#[tauri::command]
fn read_file(state: State<AppState>, generation: u64, path: String) -> Result<ReadDto, String> {
    let current = state.generation.load(Ordering::SeqCst);
    if generation != current {
        return Err("stale workspace".into());
    }
    let root = state
        .root
        .lock()
        .map_err(|e| e.to_string())?
        .clone()
        .ok_or("no workspace")?;
    match files::read_file(&root, root.join(path)).map_err(|e| format!("{e:?}"))? {
        files::ReadOutcome::Text { bytes, highlight } => Ok(ReadDto::Text { bytes, highlight }),
        files::ReadOutcome::TooLarge { bytes } => Ok(ReadDto::TooLarge { bytes }),
        files::ReadOutcome::Binary => Ok(ReadDto::Binary),
    }
}

#[tauri::command]
fn write_file(
    state: State<AppState>,
    generation: u64,
    path: String,
    text: String,
    save_generation: u64,
) -> Result<u64, String> {
    let current = state.generation.load(Ordering::SeqCst);
    if generation != current {
        return Err("stale workspace".into());
    }
    let root = state
        .root
        .lock()
        .map_err(|e| e.to_string())?
        .clone()
        .ok_or("no workspace")?;
    files::write_file(&root, &root.join(&path), &text).map_err(|e| format!("{e:?}"))?;
    let _ = save_generation;
    Ok(save_generation)
}
```

`src/shell/store.ts` holds generation, entries for the root only in this plan, tabs, active path, and status. Save ignores a result whose `saveGeneration` is older than the tab's current one.

```ts
import { invoke } from "@tauri-apps/api/core";
import { open } from "@tauri-apps/plugin-dialog";
import type { Entry, ReadDto, Tab } from "./types";

export type ShellState = {
  generation: number;
  entries: Entry[];
  tabs: Tab[];
  active: string | null;
  status: string;
};

type Listener = () => void;
let state: ShellState = {
  generation: 0,
  entries: [],
  tabs: [],
  active: null,
  status: "Open a folder",
};
const listeners = new Set<Listener>();

export function getState(): ShellState {
  return state;
}

export function subscribe(listener: Listener): () => void {
  listeners.add(listener);
  return () => listeners.delete(listener);
}

function set(next: ShellState) {
  state = next;
  listeners.forEach((l) => l());
}

function decode(bytes: number[]): string {
  return new TextDecoder("utf-8", { fatal: true }).decode(new Uint8Array(bytes));
}

export async function openFolder(): Promise<void> {
  const picked = await open({ directory: true, multiple: false });
  if (typeof picked !== "string") return;
  try {
    const opened = await invoke<{
      generation: number;
      root_entries: Entry[];
    }>("open_workspace", { path: picked });
    set({
      generation: opened.generation,
      entries: opened.root_entries,
      tabs: [],
      active: null,
      status: picked,
    });
  } catch (err) {
    set({ ...state, status: String(err) });
  }
}

export async function openPath(path: string): Promise<void> {
  const existing = state.tabs.find((t) => t.path === path);
  if (existing) {
    set({ ...state, active: path });
    return;
  }
  const entry = state.entries.find((e) => e.path === path);
  if (entry?.kind === "dir") {
    const children = await invoke<Entry[]>("list_children", {
      generation: state.generation,
      path,
    });
    set({ ...state, entries: merge(state.entries, children), status: state.status });
    return;
  }
  try {
    const read = await invoke<ReadDto>("read_file", {
      generation: state.generation,
      path,
    });
    const tab = tabFromRead(path, read);
    set({
      ...state,
      tabs: [...state.tabs, tab],
      active: path,
      status: state.status,
    });
  } catch (err) {
    set({ ...state, status: String(err) });
  }
}

function merge(current: Entry[], children: Entry[]): Entry[] {
  const paths = new Set(children.map((c) => c.path));
  return [...current.filter((e) => !paths.has(e.path)), ...children].sort((a, b) =>
    a.path.localeCompare(b.path),
  );
}

function tabFromRead(path: string, read: ReadDto): Tab {
  if (read.kind === "binary") {
    return { path, text: null, savedText: null, highlight: false, notice: { kind: "binary" }, saveGeneration: 0 };
  }
  if (read.kind === "too_large") {
    return {
      path,
      text: null,
      savedText: null,
      highlight: false,
      notice: { kind: "too_large", bytes: read.bytes },
      saveGeneration: 0,
    };
  }
  const text = decode(read.bytes);
  return {
    path,
    text,
    savedText: text,
    highlight: read.highlight,
    notice: null,
    saveGeneration: 0,
  };
}

export function edit(path: string, text: string) {
  set({
    ...state,
    tabs: state.tabs.map((t) => (t.path === path ? { ...t, text } : t)),
  });
}

export async function saveActive(): Promise<void> {
  const tab = state.tabs.find((t) => t.path === state.active);
  if (!tab || tab.text === null) return;
  const saveGeneration = tab.saveGeneration + 1;
  set({
    ...state,
    tabs: state.tabs.map((t) => (t.path === tab.path ? { ...t, saveGeneration } : t)),
  });
  try {
    const echoed = await invoke<number>("write_file", {
      generation: state.generation,
      path: tab.path,
      text: tab.text,
      saveGeneration,
    });
    const current = getState().tabs.find((t) => t.path === tab.path);
    if (!current || current.saveGeneration !== echoed) return;
    set({
      ...getState(),
      tabs: getState().tabs.map((t) =>
        t.path === tab.path && t.saveGeneration === echoed
          ? { ...t, savedText: t.text }
          : t,
      ),
      status: `Saved ${tab.path}`,
    });
  } catch (err) {
    set({ ...getState(), status: String(err) });
  }
}

export function closeTab(path: string) {
  const tabs = state.tabs.filter((t) => t.path !== path);
  set({
    ...state,
    tabs,
    active: state.active === path ? (tabs[0]?.path ?? null) : state.active,
  });
}
```
